# Tile Keys Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The selected tile is what the keyboard acts on: arrows select a neighbour, Shift+arrows and Shift+1–9 carry
the selected tile (swap, selection follows), X / Delete / Backspace clear it.

**Architecture:** `Hotkey.of` (packages/api) reads the new keys into new `Hotkey` cases. `Display.press` (apps/web) turns
them into the existing `Swap` and `Clear` screen commands plus a change of selection. "The neighbour that way" is a
pure method on `Layout`, from the slot rectangles it already has. The shell's stream views need no change.

**Tech Stack:** TypeScript 7 (tsgo), Effect 4 `Schema`, React, Vitest via `@effect/vitest`, pnpm.

**Spec:** `docs/specs/2026-11-01-tile-keys-design.md`

- [ ] **Step 3: Implement** — in `Layout.ts`, after `slots()` inside the class:

```ts
  /**
   * The slot that way from slot `from`: of the slots whose centre lies that way, those overlapping it across the axis
   * first, then the nearest along it, then the nearest across it. None at an edge.
   */
  neighbour(from: number, direction: "up" | "down" | "left" | "right"): number | undefined {
    const slots = this.slots()
    const origin = slots[from]
    if (origin === undefined) return undefined

    const [along, across] =
      direction === "left" || direction === "right" ? (["x", "y"] as const) : (["y", "x"] as const)
    const sign = direction === "right" || direction === "down" ? 1 : -1
    const ahead = slots
      .map((slot, index) => ({
        index,
        distance: (middle(slot, along) - middle(origin, along)) * sign,
        offset: Math.abs(middle(slot, across) - middle(origin, across)),
        overlaps: overlap(slot, origin, across) > 0
      }))
      .filter((candidate) => candidate.distance > 0)
      .toSorted(
        (a, b) => Number(b.overlaps) - Number(a.overlaps) || a.distance - b.distance || a.offset - b.offset
      )
    return ahead[0]?.index
  }
```

At the bottom of the file, after `rect`:

```ts
/** The middle of a slot on one axis. */
function middle(slot: Slot, axis: "x" | "y"): number {
  return axis === "x" ? slot.x + slot.width / 2 : slot.y + slot.height / 2
}

/** How far two slots overlap on one axis: zero or less when they only touch or are apart. */
function overlap(a: Slot, b: Slot, axis: "x" | "y"): number {
  const [start, size] = axis === "x" ? (["x", "width"] as const) : (["y", "height"] as const)
  return Math.min(a[start] + a[size], b[start] + b[size]) - Math.max(a[start], b[start])
}
```

(`toSorted` is stable, so equal candidates keep slot order.)

     * What a key press means, read by its physical `code`; none for a key the page should have. The arrows select,
     * or with Shift carry, the selected tile; while the guide is open, plain arrows move in it instead, and Enter,
     * the brackets, I, Delete and Backspace are the guide's.
     */
    of(key: {
      code: string
      shift: boolean
      alt: boolean
      control: boolean
      meta: boolean
      guideOpen: boolean
    }): Hotkey | undefined {
      if (key.control || key.meta) return key.code === "KeyK" ? { _tag: "CommandPalette" } : undefined
      const digit = /^Digit([1-9])$/.exec(key.code)?.[1]
      if (key.alt) return digit === undefined ? undefined : { _tag: "SelectTile", slot: Number(digit) - 1 }
      if (digit !== undefined) return { _tag: key.shift ? "Carry" : "FocusAudio", slot: Number(digit) - 1 }
      const way = arrows[key.code]
      if (way !== undefined) {
        if (key.shift) return { _tag: "CarryToward", direction: way }
        return key.guideOpen ? { _tag: "Move", direction: way } : { _tag: "SelectToward", direction: way }

      const guide = key.guideOpen ? guideKey(key.code) : undefined
      if (guide !== undefined) return guide
      switch (key.code) {
        case "KeyF":
          return { _tag: "ToggleSolo" }
        case "Escape":
          return { _tag: "Unsolo" }
        case "F11":
          return { _tag: "FullScreen" }
        case "KeyL":
          return { _tag: "NextLayout" }
        case "KeyI":
          return { _tag: "Interact" }
        case "KeyG":
          return { _tag: "Guide" }
        case "KeyX":
        case "Delete":
        case "Backspace":
          return { _tag: "ClearTile" }
        case "Slash":
          return key.shift ? { _tag: "ShowKeys" } : undefined

/** A key that is the guide's while it is open: Enter, the brackets, I, Delete and Backspace. */
function guideKey(code: string): Hotkey | undefined {
  switch (code) {
    case "Enter":
      return { _tag: "Confirm" }
    case "BracketRight":
      return { _tag: "NextEntry" }
    case "BracketLeft":
      return { _tag: "PreviousEntry" }
    case "KeyI":
      return { _tag: "Info" }
    case "Delete":
    case "Backspace":
      return { _tag: "Remove" }
    default:
      return undefined
  }
}

/** The arrow keys, by `code`, and which way each points. */
const arrows: Partial<Record<string, typeof direction.Type>> = {
  ArrowUp: "up",
  ArrowDown: "down",
  ArrowLeft: "left",
  ArrowRight: "right"
}
