# Prompts Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A toast asks whether to bring a game that is on no tile into the main tile when it enters a red zone or crunch time, can be streamed, and outranks the weakest game on screen.

**Architecture:** A pure `Prompts` fold in `apps/web/src/prompts/` (beside the autopilot) computes the one prompt from the Screen and the live state. `PromptsState` keeps it in the browser, accepts through a new `AutopilotState.yours` atom (so accepted commands are yours to the autopilot), and logs to Grafana. A Sonner toast and a top-bar toggle show it; `P`/`Y`/`N` hotkeys drive it.

**Tech Stack:** TypeScript 7, Effect 4 (`effect`, `effect/reactivity` atoms), React with `@effect/atom-react`, Sonner, Vitest with `@effect/vitest`.

Spec: `docs/specs/2026-10-10-prompts-design.md`. Worktree: `../sportal-prompts`, branch `prompts`. Run every command from the worktree root.

```ts
import type { GameId } from "@sportal/domain/Game"
import { Data, type DateTime, Option } from "effect"
import { label, type Rank } from "../autopilot/Ranking.ts"
import type { ScreenCommand } from "../screen/Screen.ts"

/**
 * A suggestion to bring `game`, on no tile, into the main tile: in `rank` (a red zone or crunch time) since `since`,
 * by way of the `target` slot, which `replaces` holds, or none when it is empty.
 */
export class Prompt extends Data.Class<{
  readonly game: GameId
  readonly rank: Rank
  readonly since: DateTime.Utc
  readonly target: number
  readonly replaces: Option.Option<{ readonly game: GameId; readonly rank: Rank }>
}> {
  /** What accepting it sends: the game into the target slot, then promoted into the main tile with the sound. */
  commands(): ReadonlyArray<ScreenCommand> {
    return [
      { _tag: "Assign", tile: this.target, game: this.game },
      { _tag: "Promote", tile: this.target }
    ]
  }

  /** It in words, each game named by `nameOf`: `PHI @ DAL — red zone, replaces CHI @ GB (on a break)`. */
  words(nameOf: (game: GameId) => string): string {
    const into = Option.match(this.replaces, {
      onNone: () => "into an empty tile",
      onSome: ({ game, rank }) => `replaces ${nameOf(game)} (${label(rank)})`
    })
    return `${nameOf(this.game)} — ${label(this.rank)}, ${into}`
  }
}
```

```ts
import type { SlateChanged } from "@sportal/api/LiveMessage"
import { GameId } from "@sportal/domain/Game"
import type { Slate } from "@sportal/domain/Slate"
import { Array as Arr, DateTime, Option, Order, Record as Rec, Schema } from "effect"
import { byRank, outranks, type Rank, rankOf } from "../autopilot/Ranking.ts"
import { Screen, Tile } from "../screen/Screen.ts"
import { Prompt } from "./Prompt.ts"

type Classifications = SlateChanged["classifications"]

/**
 * A Display's prompts: on or off, and the games you dismissed. A prompt asks to bring a game on no tile into the main
 * tile while it is in a red zone or crunch time, a source can stream it, and it has an empty slot to go to or outranks
 * the weakest game on screen. Pure.
 */
export class Prompts extends Schema.Class<Prompts>("sportal/Prompts")({
  on: Schema.Boolean,
  /** Each game you dismissed, and the `since` of the moment you dismissed it in. */
  dismissed: Schema.Record(GameId, Schema.DateTimeUtcFromString)
}) {
  /** On, nothing dismissed. */
  static readonly initial: Prompts = new Prompts({ on: true, dismissed: {} })

  /** Off when on, on when off; the dismissals kept. */
  toggle(): Prompts {
    return new Prompts({ ...this, on: !this.on })
  }

  /**
   * The prompt to show on `screen` with `live`: of the games that qualify, the best by rank, then score, then slate
   * order. None while off or soloed.
   */
  prompt(
    screen: Screen,
    live: {
      readonly slate: Slate
      readonly availability: SlateChanged["availability"]
      readonly classifications: Classifications
    }
  ): Option.Option<Prompt> {
    if (!this.on || Option.isSome(screen.solo)) return Option.none()
    const { slate, availability, classifications } = live
    const onTiles = new Set(screen.tiles.flatMap((tile) => (Tile.guards.Empty(tile) ? [] : [tile.game])))
    const candidates = slate.games.flatMap(({ snapshot: { id: game } }) => {
      const classification = classifications[game]
      if (classification === undefined || onTiles.has(game)) return []
      const rank = rankOf(classification.moment)
      const worthy = rank === "redZone" || rank === "crunchTime"
      const streams = Tile.guards.StreamTile(Screen.tileFor(game, availability[game] ?? []))
      return worthy && streams && !this.dismissedIn(game, classification.since)
        ? [{ game, rank, since: classification.since }]
        : []
    })

    return Option.flatMap(targetOn(screen, classifications), ({ tile, replaces }) => {
      const qualifying = candidates.filter(({ rank }) =>
        Option.match(replaces, { onNone: () => true, onSome: (weakest) => outranks(rank, weakest.rank) })
      )
      const best = Arr.sort(
        qualifying,
        Order.mapInput(byRank(classifications), ({ game }: { readonly game: GameId }) => ({ tile: 0, game }))
      )
      return Option.map(
        Arr.head(best),
        ({ game, rank, since }) => new Prompt({ game, rank, since, target: tile, replaces })
      )
    })
  }

  /** `prompt`'s game dismissed for its moment; each other dismissal whose game's moment has since changed is dropped. */
  dismiss(prompt: Prompt, classifications: Classifications): Prompts {
    const current = Rec.filter(this.dismissed, (since, game) => {
      const classification = classifications[game]
      return classification !== undefined && DateTime.Equivalence(classification.since, since)
    })
    return new Prompts({ ...this, dismissed: { ...current, [prompt.game]: prompt.since } })
  }

  /** Whether you dismissed `game` in the moment that began at `since`. */
  private dismissedIn(game: GameId, since: DateTime.Utc): boolean {
    const dismissed = this.dismissed[game]
    return dismissed !== undefined && DateTime.Equivalence(dismissed, since)
  }
}

/**
 * Where a prompt's game would go on `screen`: the first empty slot; with none, the slot of the weakest game on
 * screen, with that game and its rank.
 */
function targetOn(
  screen: Screen,
  classifications: Classifications
): Option.Option<{
  readonly tile: number
  readonly replaces: Option.Option<{ readonly game: GameId; readonly rank: Rank }>
}> {
  const slots = screen.onScreen()
  const empty = slots.findIndex((tile) => Tile.guards.Empty(tile))
  if (empty !== -1) return Option.some({ tile: empty, replaces: Option.none() })
  const occupied = slots.flatMap((tile, index) => (Tile.guards.Empty(tile) ? [] : [{ tile: index, game: tile.game }]))
  return Option.map(Arr.last(Arr.sort(occupied, byRank(classifications))), ({ tile, game }) => ({
    tile,
    replaces: Option.some({ game, rank: rankOf(classifications[game]?.moment) })
  }))
}
```

```ts
import { Data, Effect, Option, Stream } from "effect"
import { KeyValueStore } from "effect/persistence"
import { Atom } from "effect/reactivity"
import * as AutopilotState from "../autopilot/AutopilotState.ts"
import { nameIn } from "../autopilot/AutopilotState.ts"
import * as LiveStream from "../hub/LiveStream.ts"
import * as DisplayScreen from "../screen/DisplayScreen.ts"
import { logDefect, runtime } from "../WebRuntime.ts"
import type { Prompt } from "./Prompt.ts"
import { Prompts } from "./Prompts.ts"

/** What moves this Display's prompts: turned on or off, or the prompt accepted or dismissed. */
export type PromptInput = Data.TaggedEnum<{
  Toggled: object
  Accepted: object
  Dismissed: object
}>
export const PromptInput = Data.taggedEnum<PromptInput>()

/**
 * This Display's prompts, kept in this browser. Accepting sends the prompt's commands as yours; dismissing keeps its
 * game quiet for its moment. With no prompt, both do nothing.
 */
export const prompts: Atom.Writable<Prompts, PromptInput> = Atom.writable(
  (get) => get(kept),
  (ctx, input: PromptInput) => {
    const current = ctx.get(kept)
    if (PromptInput.$is("Toggled")(input)) {
      ctx.set(kept, current.toggle())
      return
    }
    const shown = ctx.get(prompt)
    if (Option.isNone(shown)) return
    const accepted = PromptInput.$is("Accepted")(input)
    Atom.batch(() => {
      ctx.set(answered, Option.some({ prompt: shown.value, accepted }))
      if (accepted) for (const command of shown.value.commands()) ctx.set(AutopilotState.yours, command)
      else ctx.set(kept, current.dismiss(shown.value, ctx.get(LiveStream.live).classifications))
    })
  }
).pipe(Atom.keepAlive)

/** The prompt to show now on this Display's Screen, if any. */
export const prompt: Atom.Atom<Option.Option<Prompt>> = Atom.make((get) =>
  get(kept).prompt(get(DisplayScreen.screen), get(LiveStream.live))
)

/** The last prompt you answered, and whether you accepted it; none before any. */
const answered = Atom.make<Option.Option<{ readonly prompt: Prompt; readonly accepted: boolean }>>(
  Option.none()
).pipe(Atom.keepAlive)

/** Logs each prompt shown, taken over or gone, and each you accepted or dismissed. Mounted once, by the Display. */
export const log = runtime.atom((get) =>
  Effect.gen(function* () {
    const shown = get.stream(prompt).pipe(
      Stream.changesWith((a, b) => keyOf(a) === keyOf(b)),
      Stream.zipWithPrevious,
      Stream.runForEach(([previous, next]) => {
        const nameOf = nameIn(get.once(LiveStream.live).slate)
        const before = Option.flatten(previous)
        return Option.match(next, {
          onSome: (each) =>
            Effect.logInfo(`Prompt: ${Option.isSome(before) ? "now " : ""}${each.words(nameOf)}`).pipe(
              Effect.annotateLogs(details(each))
            ),
          onNone: () => (Option.isSome(before) ? Effect.logInfo("Prompt: gone") : Effect.void)
        })
      })
    )
    const answers = get.stream(answered).pipe(
      Stream.map(Option.toArray),
      Stream.flattenIterable,
      Stream.runForEach(({ prompt: answer, accepted }) =>
        Effect.logInfo(
          `Prompt: ${accepted ? "accepted" : "dismissed"} ${answer.words(nameIn(get.once(LiveStream.live).slate))}`
        ).pipe(Effect.annotateLogs(details(answer)))
      )
    )
    yield* Effect.all([shown, answers], { concurrency: "unbounded", discard: true })
  }).pipe(logDefect("The prompts"))
)

/** What tells one prompt from another for the log: its game, rank and target. */
function keyOf(prompt: Option.Option<Prompt>): string {
  return Option.match(prompt, {
    onNone: () => "",
    onSome: ({ game, rank, target }) => `${game} ${rank} ${target}`
  })
}

/** A prompt's log annotations. */
function details(prompt: Prompt) {
  return { "prompt.game": prompt.game, "prompt.rank": prompt.rank, "prompt.target": prompt.target }
}

const kept = Atom.kvs({
  runtime: Atom.runtime(KeyValueStore.layerStorage(() => localStorage)),
  key: "sportal/prompts",
  schema: Prompts,
  defaultValue: () => Prompts.initial
}).pipe(Atom.keepAlive)
```

```ts
  const answer = useAtomSet(PromptsState.prompts)

      dismissible: false,
      action: { label: "Watch · Y", onClick: () => answer(PromptInput.Accepted()) },
      cancel: { label: "Skip · N", onClick: () => answer(PromptInput.Dismissed()) }
    })
```
