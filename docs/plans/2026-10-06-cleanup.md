```ts
/** What one step did to one game, for telemetry to tell. */
export type FoldReport = Data.TaggedEnum<{
  Took:     { snapshot: GameSnapshot; from: DeliverySource; outcome: "taken" | "behind" | "kept"; had: Option<GameSnapshot> }
  Started:  { liveGame: LiveGame; at: DateTime.Utc }
  Changed:  { before: LiveGame; after: LiveGame; at: DateTime.Utc }
  Stopped:  { liveGame: LiveGame; at: DateTime.Utc }
  Happened: { snapshot: GameSnapshot; event: GameEvent }
  Dropped:  { game: GameId; reason: "no game" }
  Raised:   { snapshot: GameSnapshot; cause: Cause.Cause<never>; by: string }
}>
export const FoldReport: Data.TaggedEnum.Constructor<FoldReport>  // + $match

/** The fold: the last frame's seq, each league's last listing, every current game's live game, and how far it has ticked. */
export class LiveGames extends Data.Class<{
  readonly seq: Option.Option<number>
  readonly listed: ReadonlyMap<LeagueId, ReadonlyArray<GameSnapshot>>
  readonly live: ReadonlyMap<GameId, LiveGame>
  readonly reached: Option.Option<DateTime.Utc>
}> {
  static readonly empty: LiveGames
  /** Ticks to `at`, then applies `delivery` if any. A step that raises a defect keeps the game's last good live game and reports `Raised`. */
  step(delivery: Option.Option<Delivery>, at: DateTime.Utc, seq: Option.Option<number>):
    Effect.Effect<{ readonly next: LiveGames; readonly reports: ReadonlyArray<FoldReport> }>
  /** The earliest instant after `reached` at which a game's tick or currency may change. */
  earliestDue(): Option.Option<DateTime.Utc>
  /** Each listed game in `leagues`, leagues in order, as the fold has it. */
  slate(leagues: ReadonlyArray<LeagueId>): ReadonlyArray<GameSnapshot>
  /** Every listed game as the fold has it, then live games no listing holds. */
  games(): ReadonlyArray<GameSnapshot>
  /** A game's watchability: its live game's, or fresh at `now` for a game the fold does not hold. */
  watchabilityOf(game: GameSnapshot, now: DateTime.Utc): Watchability
  /** Whether `that` publishes the same: same maps by reference, same seq. `reached` is ignored. */
  publishesSameAs(that: LiveGames): boolean
}
```
