# Games API and a sport-neutral model Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the one `/api/live` firehose with filterable, per-view queries (each a `GET` plus a `/live` SSE twin), and
reshape the domain so games, situations and plays are sport-neutral, with football the only sport on the new seam.

**Architecture:** Three phases, each one commit that passes `pnpm check`. Phase 1 deletes the classification. Phase 2
reshapes the domain (`League`, `Situation`, `Play`, `GameSnapshot`) and the ESPN adapter, keeping today's wire message.
Phase 3 adds the `games` API group, a `Schedule` service in the Hub that answers it (today live from the Director,
other days fetched from the provider), and moves the web onto one atom family per query, deleting the bottom bar.

**Tech Stack:** TypeScript 7 (tsgo + `@effect/tsgo`), Effect 4 (`effect/http-api`, `effect/reactivity` atoms), React,
Vitest via `@effect/vitest`, pnpm workspaces.

**Spec:** `docs/specs/2026-10-15-games-api-design.md`.

```ts
  Director,
  {
    /** The slate: the current one, then each one a take makes. */
    readonly slate: Stream.Stream<Slate>
  }
>()("sportal/hub/Director") {

      const slate = yield* SubscriptionRef.make(Slate.empty)
      const serial = yield* Semaphore.make(1)

      const take = Effect.fn("Director.take")(
        function* (delivery: Delivery) {

          const [next, changes] = Delivery.$match(delivery, {
            Listing: ({ league, games }) => before.relisted(league, games),
            Plays: ({ game, plays }) => before.playsTaken(game, plays)
          })
          if (next === before) return

          yield* SubscriptionRef.set(slate, next)
          yield* observed(changes)
        },
        serial.withPermits(1),
        Effect.catchDefect((defect) =>
          Effect.logError(
            "Director: a take raised a defect; what it had not yet set is unchanged",
            Cause.die(defect)
          )
        )
      )

        provider.deliveries.pipe(
          Stream.runForEach(take),
          loggedUnlessStopping(
            "Director: the provider's frames stopped; the slate stays as it is until the Hub restarts",
            "the provider's frames ended"
          )
        ),
        { startImmediately: true }
      )

      return Director.of({ slate: SubscriptionRef.changes(slate) })
    })
```

```ts
import { Option, Order, Schema } from "effect"
import { TeamId } from "./Team.ts"

/** Live football state: the team with the ball, the down and distance, where the ball is, and each side's timeouts. */
export class FootballSituation extends Schema.TaggedClass<FootballSituation>()("Football", {
  possession: Schema.optional(TeamId),
  down: Schema.optional(Schema.Int),
  distance: Schema.optional(Schema.Int),
  /** Yards to the goal the team with the ball attacks; absent when no team has the ball. */
  yardLine: Schema.optional(Schema.Int),
  homeTimeouts: Schema.Int,
  awayTimeouts: Schema.Int
}) {
  /** The ball is at the opponent's 20 or closer. */
  inRedZone(): boolean {
    return this.yardLine !== undefined && this.yardLine <= redZone
  }

  /** The line to gain is the goal line. */
  goalToGo(): boolean {
    return this.yardLine !== undefined && this.distance !== undefined && this.distance >= this.yardLine
  }

  /** Where the ball is in `game`: the spot of the team with the ball, at its yard line. */
  ball(game: {
    readonly home: { readonly team: { readonly id: TeamId } }
    readonly away: { readonly team: { readonly id: TeamId } }
  }): Option.Option<BallSpot> {
    return BallSpot.of(game, { team: this.possession, yardsToEndzone: this.yardLine })
  }
}

/** Live baseball state: the half inning, the outs, the count, and the runners on base. */
export class BaseballSituation extends Schema.TaggedClass<BaseballSituation>()("Baseball", {
  half: Schema.Literals(["top", "bottom"]),
  outs: Schema.Int,
  balls: Schema.Int,
  strikes: Schema.Int,
  onFirst: Schema.Boolean,
  onSecond: Schema.Boolean,
  onThird: Schema.Boolean
}) {}

const cases = Schema.Union([FootballSituation, BaseballSituation]).pipe(Schema.toTaggedUnion("_tag"))

/**
 * A live game's state, by its sport: a tagged union of its cases with `.match`, `.guards`, `.cases`, and `order`.
 * Basketball has no case until its provider sends something worth showing.
 */
export const Situation = Object.assign(cases, {
  /**
   * Later within the same period and clock: in baseball the bottom half after the top, then more outs. Any other
   * pair ties, and so does a missing situation.
   */
  order: Order.combine(
    Order.mapInput(Order.Number, (situation: typeof cases.Type | undefined) =>
      baseballOf(situation)?.half === "bottom" ? 1 : 0
    ),
    Order.mapInput(Order.Number, (situation: typeof cases.Type | undefined) => baseballOf(situation)?.outs ?? 0)
  )
})
export type Situation = typeof Situation.Type

/** Where a spot is on the field: yards from the home goal line (0–100), and the way its team is going. */
export class BallSpot extends Schema.Class<BallSpot>("sportal/BallSpot")({
  yards: Schema.Int.check(Schema.isBetween({ minimum: 0, maximum: 100 })),
  toward: Schema.Literals(["home", "away"])
}) {
  /**
   * Where a spot is in `game`, from the team with the ball and the yards it has to the goal it attacks. None when
   * either is missing, the team plays in neither side, or the yards are off the field.
   */
  static of(
    game: {
      readonly home: { readonly team: { readonly id: TeamId } }
      readonly away: { readonly team: { readonly id: TeamId } }
    },
    spot: { readonly team?: TeamId | undefined; readonly yardsToEndzone?: number | undefined }
  ): Option.Option<BallSpot> {
    const { team, yardsToEndzone } = spot
    if (team === undefined || yardsToEndzone === undefined) return Option.none()
    if (team === game.home.team.id) return BallSpot.makeOption({ yards: 100 - yardsToEndzone, toward: "away" })
    if (team === game.away.team.id) return BallSpot.makeOption({ yards: yardsToEndzone, toward: "home" })
    return Option.none()
  }
}

/** Yards to the goal at which a drive is in the red zone. */
const redZone = 20

function baseballOf(situation: Situation | undefined): BaseballSituation | undefined {
  return situation !== undefined && cases.guards.Baseball(situation) ? situation : undefined
}
```

```ts
  /** Its plays before sequence `before` (all of them without it), newest first, at most `limit`, and whether more remain. */
  playsPage(before: number | undefined, limit: number): PlaysPage {
    const earlier = before === undefined ? this.plays : this.plays.filter((play) => play.sequence < before)
    return { plays: earlier.slice(-limit).toReversed(), more: earlier.length > limit }
  }

  /**
   * What changed in its plays since `earlier`, the same game before: none when they are the same; the plays added
   * when `earlier`'s are all still first and unchanged; otherwise a reset to its latest page.
   */
  playsChange(earlier: Game): Option.Option<PlaysChange> {
    if (Equal.equals(earlier.plays, this.plays)) return Option.none()

    const kept = this.plays.slice(0, earlier.plays.length)
    if (this.plays.length > earlier.plays.length && Equal.equals(kept, earlier.plays)) {
      return Option.some({ _tag: "Appended", plays: this.plays.slice(earlier.plays.length).toReversed() })
    }
    return Option.some({ _tag: "Reset", plays: this.playsPage(undefined, Game.pageSize).plays })
  }
```

`limit` is at least 1 wherever it comes from (the API's query schema checks it), so `slice(-limit)` is safe.

```ts
import type { LeagueId } from "@sportal/domain/League"
import { Array as Arr, Data, Option } from "effect"

/**
 * A league ESPN serves the Hub: where its site scoreboard and game summary are, and its FastCast game topic prefix.
 * A sport joins by a row in `followed` and its situation and play-detail decode in `EspnEvent` and `EspnPlay`.
 */
export class EspnLeague extends Data.Class<{
  readonly league: LeagueId
  /** ESPN's sport and league in its URLs, as `football/nfl`. */
  readonly path: string
  /** What the scoreboard needs besides a date, as `groups=80&limit=300`; empty for none. */
  readonly scoreboardQuery: string
  readonly topicPrefix: string
}> {
  static readonly followed: ReadonlyArray<EspnLeague> = [
    new EspnLeague({ league: "nfl", path: "football/nfl", scoreboardQuery: "", topicPrefix: "gp-football-nfl-" }),
    new EspnLeague({
      league: "fbs",
      path: "football/college-football",
      scoreboardQuery: "groups=80&limit=300",
      topicPrefix: "gp-football-college-football-"
    })
  ]

  /** League `league`, when ESPN serves it to the Hub. */
  static of(league: LeagueId): Option.Option<EspnLeague> {
    return Arr.findFirst(EspnLeague.followed, (each) => each.league === league)
  }

  /** The league and game of FastCast topic `tc`, when it is a followed league's game topic. */
  static ofTopic(tc: string): Option.Option<{ readonly league: EspnLeague; readonly game: GameId }> {
    return Arr.findFirst(EspnLeague.followed, (league) =>
      tc.startsWith(league.topicPrefix)
        ? Option.some({ league, game: GameId.make(tc.slice(league.topicPrefix.length)) })
        : Option.none()
    )
  }

  /** The site scoreboard: today's games (every FBS game of the week for college), or day `date`'s (`YYYY-MM-DD`). */
  scoreboardUrl(date: Option.Option<string>): string {
    const query = [
      ...(this.scoreboardQuery === "" ? [] : [this.scoreboardQuery]),
      ...Option.toArray(Option.map(date, (day) => `dates=${day.replaceAll("-", "")}`))
    ].join("&")
    return `${base}/${this.path}/scoreboard${query === "" ? "" : `?${query}`}`
  }

  /** The game summary of `game`, which carries its plays. */
  summaryUrl(game: GameId): string {
    return `${base}/${this.path}/summary?event=${game}`
  }

  /** FastCast's game topic of `game`. */
  topicOf(game: GameId): string {
    return `${this.topicPrefix}${game}`
  }
}
```

```ts
import { GameSnapshot, Progress } from "@sportal/domain/Game"
import { LeagueId } from "@sportal/domain/League"
import { Order, Schema, SchemaGetter } from "effect"

/** A calendar day as `YYYY-MM-DD`. */
export const CalendarDate = Schema.String.check(Schema.isPattern(/^\d{4}-\d{2}-\d{2}$/)).pipe(
  Schema.brand("CalendarDate")
)
export type CalendarDate = typeof CalendarDate.Type

/** A day's games, as a query answers them: the day, and its games filtered and in watch order. */
export const GameList = Schema.Struct({ date: CalendarDate, games: Schema.Array(ListedGame) })
export type GameList = typeof GameList.Type

export const GameListJson = Schema.fromJsonString(GameList)

/**
 * What a list of games asks for, as its query string spells it: a day (today in the Hub's time zone without one),
 * leagues and progress as comma lists (all without them), and `carried=true` for only the games a source carries or
 * will carry. With `answer`, which filters and orders.
 */
export const GamesQuery = Object.assign(
  Schema.Struct({
    date: Schema.optional(CalendarDate),
    league: Schema.optional(commaList(LeagueId)),
    progress: Schema.optional(commaList(Progress)),
    carried: Schema.optional(Schema.Literal("true"))
  }),
  { answer }
)
export type GamesQuery = typeof GamesQuery.Type

/** `games` that `query` keeps (all but its day, which picks the games before this), live first, then upcoming, then over. */
function answer(
  query: Omit<typeof GamesQuery.Type, "date">,
  games: ReadonlyArray<ListedGame>
): ReadonlyArray<ListedGame> {
  return games
    .filter(
      ({ snapshot, availability }) =>
        (query.league === undefined || query.league.includes(snapshot.league)) &&
        (query.progress === undefined || query.progress.includes(snapshot.progress())) &&
        (query.carried === undefined || availability.some(carriedOrUpcoming))
    )
    .toSorted(byWatchOrder)
}

/** A comma list in a query string, as `nfl,fbs`, read as the values of `item`. */
function commaList<S extends Schema.Top & { readonly Encoded: string }>(item: S) {
  return Schema.String.pipe(
    Schema.decodeTo(Schema.Array(item), {
      decode: SchemaGetter.transform((list: string) => list.split(",")),
      encode: SchemaGetter.transform((items: ReadonlyArray<string>) => items.join(","))
    })
  )
}

function carriedOrUpcoming(availability: ChannelAvailability): boolean {
  return ChannelAvailability.guards.Carried(availability) || ChannelAvailability.guards.Upcoming(availability)
}

const byWatchOrder = Order.mapInput(GameSnapshot.watchOrder, ({ snapshot }: ListedGame) => snapshot)
```

```ts
import { CalendarDate, GameList, GamesQuery } from "@sportal/api/GamesQuery"
import { GameNotFound } from "@sportal/api/GameNotFound"
import { ListedGame } from "@sportal/api/ListedGame"
import { ScheduleUnavailable } from "@sportal/api/ScheduleUnavailable"
import { Game, type GameId, type GameSnapshot, type PlaysChange, type PlaysPage } from "@sportal/domain/Game"
import { Play } from "@sportal/domain/Play"
import {
  Cache,
  Context,
  DateTime,
  Duration,
  Effect,
  Exit,
  HashMap,
  Layer,
  Option,
  Ref,
  Schema,
  Sink,
  Stream
} from "effect"
import { Director } from "./Director.ts"
import { HubConfig } from "./HubConfig.ts"
import { Provider } from "./provider/Provider.ts"
import { Source } from "./source/Source.ts"

/**
 * What the games API answers: a day's games, one game, and a game's plays, each once or live. Today is the
 * Director's slate, live; another day is the provider's listing for it, held in memory (a past day until the Hub
 * restarts, a later one for 10 minutes). A game off today's slate is found through a day already listed, and its plays
 * are fetched from the provider. Each live answer is one connection's own stream, sending only what changed.
 */
export class Schedule extends Context.Service<
  Schedule,
  {
    readonly list: (query: GamesQuery) => Effect.Effect<GameList, ScheduleUnavailable>
    readonly listLive: (query: GamesQuery) => Effect.Effect<Stream.Stream<GameList>, ScheduleUnavailable>
    readonly game: (id: GameId) => Effect.Effect<ListedGame, GameNotFound | ScheduleUnavailable>
    readonly gameLive: (
      id: GameId
    ) => Effect.Effect<Stream.Stream<ListedGame>, GameNotFound | ScheduleUnavailable>
    readonly plays: (
      id: GameId,
      page: { readonly before?: number | undefined; readonly limit?: number | undefined }
    ) => Effect.Effect<PlaysPage, GameNotFound | ScheduleUnavailable>
    readonly playsLive: (
      id: GameId
    ) => Effect.Effect<Stream.Stream<PlaysChange>, GameNotFound | ScheduleUnavailable>
  }
>()("sportal/hub/Schedule") {
  static readonly layer: Layer.Layer<Schedule, never, Director | Provider | Source | HubConfig> = Layer.effect(
    Schedule,
    Effect.gen(function* () {
      const director = yield* Director
      const provider = yield* Provider
      const source = yield* Source
      const config = yield* HubConfig
      const zone = yield* DateTime.zoneMakeNamedEffect(config.timeZone).pipe(Effect.orDie)
      // Each game a fetched day listed, by id, so a game off today's slate can be found.
      const held = yield* Ref.make(HashMap.empty<GameId, GameSnapshot>())

      const today = Effect.map(DateTime.now, (now) => dayOf(now, zone))

      const days = yield* Cache.make({
        capacity: 64,
        lookup: Effect.fnUntraced(function* (date: CalendarDate) {
          const games = yield* provider.listing(date).pipe(Effect.mapError(() => new ScheduleUnavailable()))
          const now = yield* today

          yield* Ref.update(held, (all) => HashMap.setMany(all, games.map((game) => [game.id, game] as const)))
          return { games, past: date < now }
        }),
        timeToLive: (exit) =>
          Exit.isSuccess(exit) ? (exit.value.past ? Duration.infinity : Duration.minutes(10)) : Duration.zero
      })

      const plays = yield* Cache.make({
        capacity: 256,
        lookup: (snapshot: GameSnapshot) =>
          provider.plays(snapshot.league, snapshot.id).pipe(Effect.mapError(() => new ScheduleUnavailable())),
        timeToLive: (exit, snapshot) =>
          Exit.isSuccess(exit) && snapshot.progress() === "over" ? Duration.infinity : Duration.minutes(1)
      })

      /** Day `date`'s games, the current ones then each change: today's follow the slate, another day's never change. */
      const gamesOn = Effect.fnUntraced(function* (date: CalendarDate) {
        const now = yield* today
        if (date === now) {
          return director.slate.pipe(
            Stream.map((slate) =>
              slate.games.filter(
                ({ snapshot }) => dayOf(snapshot.startsAt, zone) === date || snapshot.progress() === "live"
              )
            )
          )
        }
        const listed = yield* Cache.get(days, date)

        return Stream.make(listed.games.map(Game.of))
      })

      /** Game `id`, the current one then each change: from today's slate, or once from a held day with its plays. */
      const gameOf = Effect.fnUntraced(function* (id: GameId) {
        const slate = yield* first(director.slate)
        if (Option.isSome(slate.gameOf(id))) {
          return director.slate.pipe(
            Stream.map((each) => each.gameOf(id)),
            Stream.filter(Option.isSome),
            Stream.map(({ value }) => value),
            Stream.changes
          )
        }
        const all = yield* Ref.get(held)
        const snapshot = yield* Option.match(HashMap.get(all, id), {
          onNone: () => Effect.fail(new GameNotFound()),
          onSome: Effect.succeed
        })
        const fetched = yield* Cache.get(plays, snapshot)

        return Stream.make(new Game({ snapshot, plays: fetched.toSorted(Play.order) }))
      })

      /** `games` beside each one's availability now; rechecked every 30 seconds while the stream runs. */
      const listed = (games: Stream.Stream<ReadonlyArray<Game>>): Stream.Stream<ReadonlyArray<ListedGame>> =>
        Stream.zipLatest(games, Stream.tick("30 seconds")).pipe(
          Stream.mapEffect(([current]) =>
            Effect.gen(function* () {
              const availabilityOf = yield* source.availability
              const now = yield* DateTime.now

              return current.map(({ snapshot }) => ({ snapshot, availability: availabilityOf(snapshot, now) }))
            })
          )
        )

      const listLive = Effect.fn("Schedule.listLive")(function* (query: GamesQuery) {
        const now = yield* today
        const date = query.date ?? now
        const games = yield* gamesOn(date)

        return settled(
          listed(games).pipe(
            Stream.map((each): GameList => ({ date, games: GamesQuery.answer(query, each) })),
            Stream.changesWith(sameList)
          )
        )
      })

      const gameLive = Effect.fn("Schedule.gameLive")(function* (id: GameId) {
        const game = yield* gameOf(id)

        return listed(Stream.map(game, (each) => [each])).pipe(
          Stream.map((each) => each[0]),
          Stream.filter((each) => each !== undefined),
          Stream.changesWith(sameListed)
        )
      })

      const playsLive = Effect.fn("Schedule.playsLive")(function* (id: GameId) {
        const game = yield* gameOf(id)

        return game.pipe(
          Stream.mapAccum(
            () => Option.none<Game>(),
            (earlier, next) => {
              const change = Option.match(earlier, {
                onNone: () =>
                  Option.some<PlaysChange>({ _tag: "Reset", plays: next.playsPage(undefined, Game.pageSize).plays }),
                onSome: (had) => next.playsChange(had)
              })
              return [Option.some(next), Option.toArray(change)] as const
            }
          )
        )
      })

      return Schedule.of({
        list: Effect.fn("Schedule.list")(function* (query: GamesQuery) {
          const answers = yield* listLive(query)

          return yield* first(answers)
        }),
        listLive,
        game: Effect.fn("Schedule.game")(function* (id: GameId) {
          const answers = yield* gameLive(id)

          return yield* first(answers)
        }),
        gameLive,
        plays: Effect.fn("Schedule.plays")(function* (id: GameId, page) {
          const game = yield* gameOf(id)
          const current = yield* first(game)

          return current.playsPage(page.before, page.limit ?? Game.pageSize)
        }),
        playsLive
      })
    })
  )
}

/** The calendar day `at` falls on in `zone`, as `YYYY-MM-DD`. */
function dayOf(at: DateTime.DateTime, zone: DateTime.TimeZone): CalendarDate {
  return CalendarDate.make(DateTime.formatIsoDate(DateTime.setZone(at, zone)))
}

/** The first value of `stream`, which always has one: a live answer starts with the current one. */
function first<A>(stream: Stream.Stream<A>): Effect.Effect<A> {
  return Stream.runHead(stream).pipe(
    Effect.flatMap(
      Option.match({
        onNone: () => Effect.die("Schedule: a live answer ended before its first value"),
        onSome: Effect.succeed
      })
    )
  )
}

/** `changes` with its first value at once and each later one after 250 ms with no other. */
function settled<A>(changes: Stream.Stream<A>): Stream.Stream<A> {
  return Stream.unwrap(
    Effect.map(Stream.peel(changes, Sink.head()), ([current, rest]) =>
      Stream.concat(Stream.fromIterable(Option.toArray(current)), rest.pipe(Stream.debounce("250 millis")))
    )
  )
}

const sameList = Schema.toEquivalence(GameList)

const sameListed = Schema.toEquivalence(ListedGame)
```

```ts
 * The live stream at `url`, its messages folded by `fold` from `initial`, and whether it is up. The EventSource opens
 * when the atom is first read and closes when nothing reads it; it reconnects after an error, and the Hub sends the
 * current answer on every connection. A message that does not decode is logged and skipped.
 */
export function of<M, S>(
  url: string,
  message: Schema.Codec<M, string>,
  initial: S,
  fold: (state: S, message: M) => S
): Atom.Atom<Live<S>> {
  const decode = Schema.decodeUnknownEffect(message)
  const start: Live<S> = { connection: Connection.Connecting(), value: initial }
  const step = (current: Live<S>, event: ServerEvent): Effect.Effect<Live<S>> =>
    ServerEvent.$match(event, {
      Opened: () => Effect.succeed({ ...current, connection: Connection.Connected() }),
      // Only a stream that was up is lost; one that never connected is still connecting, and one already lost
      // keeps the instant it was lost.
      Errored: () =>
        Connection.$is("Connected")(current.connection)
          ? Effect.map(DateTime.now, (since) => ({ ...current, connection: Connection.Lost({ since }) }))
          : Effect.succeed(current),
      Received: ({ data }) =>
        decode(data).pipe(
          Effect.map((each) => ({ ...current, value: fold(current.value, each) })),
          Effect.catchTag("SchemaError", (error) =>
            Effect.logWarning(`Skipped a message on ${url} that did not decode`, error).pipe(Effect.as(current))
          )
        )
    })
  return Atom.make(ServerEvents.from(url).pipe(Stream.scanEffect(start, step))).pipe(
    Atom.map(AsyncResult.getOrElse(() => start))
  )
}

/** A live stream's folded messages, and whether it is up. */
export interface Live<S> {
  readonly connection: Connection
  readonly value: S
}
```
